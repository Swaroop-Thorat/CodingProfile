import java.lang.*;
import java.util.*;

public class ACandiesForNephews {
    static Scanner sc = new Scanner(System.in);

    public static void main(String[] args) {
        int t = sc.nextInt();
        while (t-- > 0) {
            solve();
        }
    }

    static void solve() {
        int n=sc.nextInt();
        int rem=n%3;
        int needed=(3-rem)%3;
        System.out.println(needed);
    }
}
