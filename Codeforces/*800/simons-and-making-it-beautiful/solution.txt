import java.util.*;

public class Main{
    public static void main(String[] args){
        Scanner sc=new Scanner(System.in);
        int x=sc.nextInt();
        for(int i=0;i<x;i++){
            int num=sc.nextInt();
            int max=0,idx=-1;
            int[] arr=new int[num];
            for(int j=0;j<num;j++){
                arr[j]=sc.nextInt();
                if(arr[j]>max){
                    max=arr[j];
                    idx=j;
                }
            }
            int temp=arr[idx];
            arr[idx]=arr[0];
            arr[0]=temp;
            for(int n:arr){
              System.out.print(n+" ");
            }
            System.out.println();
        }
    }
}
