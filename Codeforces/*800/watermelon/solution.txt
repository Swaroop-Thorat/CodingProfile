import java.lang.*;
import java.util.*;

public class AWatermelon {
    static Scanner sc = new Scanner(System.in);

    public static void main(String[] args) {
        int n = sc.nextInt();
        if(n%2==0 && n!=2){
            System.out.println("YES");
        }
        else{
            System.out.println("NO");
        }
    }

    
}
