import java.lang.*;
import java.util.*;

public class ATeam {
    static Scanner sc = new Scanner(System.in);

    public static void main(String[] args) {
        int t = sc.nextInt();
        int res=0;
        while (t-- > 0) {
            res+=solve();
        }
        System.out.println(res);
    }

    static int solve() {
        int[] arr=new int[3];
        int count=0;
        for(int i=0;i<3;i++){
            arr[i]=sc.nextInt();
            if(arr[i]==1) count++;
        }
        if(count==2 || count==3){
            return 1;
        }
        else{
            return 0;
        }
    }
}
