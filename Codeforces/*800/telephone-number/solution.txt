import java.util.*;
import java.lang.*;
import java.io.*;

public class Codechef
{
    static Scanner sc = new Scanner(System.in);
    
	public static void main (String[] args) throws java.lang.Exception
	{
	    int t = sc.nextInt();
        while (t-- > 0) {
            solve();
        }

	}
    static void solve() {
        int n=sc.nextInt();
        String str=sc.next();
        
        if(n<11){
            System.out.println("NO");
        }
        else if(n == 11 && str.charAt(0)=='8'){
            System.out.println("YES");
        }
        else{
            int i=0,lim=n-10;
            while(i<lim){
                if(str.charAt(i)=='8'){
                    System.out.println("YES");
                    return;
                }
                i++;
            }
            System.out.println("NO");
        }
    }
}
