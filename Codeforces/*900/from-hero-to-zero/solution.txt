import java.util.*;
import java.lang.*;
import java.io.*;

public class Codechef
{
	static Scanner sc = new Scanner(System.in);
	
	public static void main (String[] args) throws java.lang.Exception
	{
        int t = sc.nextInt();
        while(t-- > 0){
            solve();
        }
    }
    
    static void solve(){
        long n=sc.nextLong();
        long k=sc.nextLong();
        long res=0;
        while(n>0){
            long rem=n%k;
            res+=rem;
            if(n>=k){
                res+=1;
            }
            n/=k;
        }
        
        System.out.println(res);
    }

}
