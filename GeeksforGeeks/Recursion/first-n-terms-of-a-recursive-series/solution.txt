class Solution {
    public ArrayList<Integer> gfSeries(int n) {
        // code here
        ArrayList<Integer> list=new ArrayList<>();
        
        for(int i=0;i<n;i++){
            list.add(0);
        }
        if (n>=2) list.set(1, 1);
        rec(list,n-1);
        return list;
    }
    static void rec(ArrayList<Integer> list,int n){
        if(n<2){
            return;
        }
        rec(list,n-1);
        
        int n2=list.get(n-1);
        int n1=list.get(n-2);
        
        int curr=(n1*n1)-n2;
        list.set(n,curr);
        
    }
}
