class Solution {
    public int reverseExponentiation(int n) {
        // code here
        String temp = new StringBuilder(String.valueOf(n)).reverse().toString();
        return f(Integer.parseInt(temp),n);
    }
    static int f(int p,int b){
        if(p==1) return b;
        
        int pow=f(p/2,b)*f(p/2,b);
        if(p%2!=0) pow*=b;
        return pow;
    }
}
