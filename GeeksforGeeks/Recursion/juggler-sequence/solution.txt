class Solution {
    static List<Long> jugglerSequence(long n) {
        // code here
        List<Long> list=new ArrayList<>();
        rec(list,n);
        return list;
    }
    static void rec(List<Long> list,long n){
        if(n==1){
            list.add(n);
            return;
        }
        list.add(n);
        if(n%2==0){
            long num=(long)Math.sqrt(n);
            rec(list,num);
        }
        else{
            long num=(long)Math.pow(n,1.5);
            rec(list,num);
        }
    }
}
