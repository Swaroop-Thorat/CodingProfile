class Solution {
    static final int MOD=(int) 1e9+7;
    public int sequence(int n) {
        // code here
        return rec(n);
    }
    static int rec(int n){
        if(n==1){
            return 1;
        }
        int ret=rec(n-1);
        long ele = ((long)(n - 1) * n) / 2 + 1; 
        long mul=1;
        for(int i=0;i<n;i++){
            mul=(mul*ele)%MOD;
            ele++;
        }
        return (int)(ret+mul)%MOD;
    }
};
