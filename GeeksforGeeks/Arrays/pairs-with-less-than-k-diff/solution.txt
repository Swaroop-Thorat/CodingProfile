class Solution {
    public static int countPairs(int arr[], int k) {
        // code here
        Arrays.sort(arr);
        int count=0;
        for(int i=0;i<arr.length;i++){
            int low=i+1,high=arr.length-1;
            int pos=i;
            while(low<=high){
                int mid=low+(high-low)/2;
                if(Math.abs(arr[i]-arr[mid])<k){
                    low=mid+1;
                    pos=mid;
                }
                else{
                    high=mid-1;
                }
            }
            count+=(pos-i);
        }
        return count;
    }
}
