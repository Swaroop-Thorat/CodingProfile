class Solution {
    static class Pair{
        Node node;
        int dist;
        Pair(Node node,int dist){
            this.dist=dist;
            this.node=node;
        }
    }
    public ArrayList<Integer> topView(Node root) {
        // code here
        SortedMap<Integer,Integer> map=new TreeMap<>();
        bfs(root,map);
        ArrayList<Integer> list=new ArrayList();
        for(int key:map.keySet()){
            list.add(map.get(key));
        }
        return list;
    }
    static void bfs(Node node,SortedMap<Integer,Integer> map){
        if(node==null) return;
        Queue<Pair> q=new ArrayDeque<>();
        q.offer(new Pair(node,0));
        while(!q.isEmpty()){
                Pair pair=q.poll();
                Node curr=pair.node;
                int dist=pair.dist;
                if(!map.containsKey(dist)){
                    map.put(dist,curr.data);
                }
                
                if(curr.left!=null) q.offer(new Pair(curr.left,dist-1));
                if(curr.right!=null) q.offer(new Pair(curr.right,dist+1));
        }
    }
}
