/* Structure of Binary Tree Node
class Node {
    int data;
    Node left;
    Node right;

    Node(int data) {
        this.data = data;
        left = null;
        right = null;
    }
}
}*/
class Solution {
    public int longestConsecutive(Node root) {
        // code here
        return dfs(root,root.data-1,0);
    }
    static int dfs(Node node,int parentVal,int currLen){
        if(node==null){
            return currLen;
        }
        
        if(parentVal==node.data-1){
            currLen++;
        }
        else{
            currLen=1;
        }
        
        int left=dfs(node.left,node.data,currLen);
        int right=dfs(node.right,node.data,currLen);
        
        return Math.max(left,right);
    }
}
