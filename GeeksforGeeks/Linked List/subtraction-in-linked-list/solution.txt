/*

Definition for singly Link List Node
class Node
{
    int data;
    Node next;

    Node(int x){
        data = x;
        next = null;
    }
}

You can also use the following for printing the link list.
Node.printList(Node node);
*/

class Solution {
    
    static int getSize(Node head){
        int size=0;
        while(head!=null){
            size++;
            head=head.next;
        }
        return size;
    }
    
    static int compare(Node head1,Node head2){
        while(head1!=head2){
            if(head1.data>head2.data){
                return 1;
            }
            else if(head2.data>head1.data){
                return 2;
            }
            head1=head1.next;
            head2=head2.next;
        }
        return 1;
    }
    
    static Node rev(Node head){
        if(head.next==null) return head;
        
        Node prev=null;
        Node nxt=null;
        while(head!=null){
            nxt=head.next;
            head.next=prev;
            prev=head;
            head=nxt;
        }
        return prev;
    }
    
    static Node sub(Node big,Node small){
        Node head=null;
        Node curr=null;
        int borrow=0;
        while(big!=null || small!=null){
            int n1=(big==null)?0:big.data;
            int n2=(small==null)?0:small.data;
            
            if(borrow!=0) {
                n1-=borrow;
                borrow=0;
            }
            if(n1<n2){
                n1+=10;
                borrow=1;
            }
            int ans=n1-n2;
            Node newNode=new Node(ans);
            
            if(head==null){
                head=newNode;
                curr=newNode;
            }
            else{
                curr.next=newNode;
                curr=newNode;
            }
            if(big!=null)big=big.next;
            if(small!=null)small=small.next;
        }
        return head;
    }
    
    static Node clean(Node head){
        while(head!=null && head.next!=null && head.data==0){
            Node temp=head;
            head=head.next;
            temp=null;
        }
        return head;
    }
    
    static Node subLinkedList(Node head1, Node head2) {
        // code here
        int size1=getSize(head1);
        int size2=getSize(head2);
        
        Node big=null;
        Node small=null;
        int res=1;
        if(size1==size2){
         res=compare(head1,head2);
        }
        
        if(size1<size2 || res==2){
            big=head2;
            small=head1;
        }
        else{
            big=head1;
            small=head2;
        }
        
        big=rev(big);
        small=rev(small);
        Node ans=sub(big,small);
        ans=rev(ans);
        ans=clean(ans);
        
        return ans;
    }
}
