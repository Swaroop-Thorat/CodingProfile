class Solution {
    public Node segregate(Node head) {
        // code here
        
        if(head.next==null) return head;
        
        Node oneHead=null , twoHead=null, zeroHead=null;
        Node one=null , two=null, zero=null;
        
        while(head!=null){
            if(head.data==1){
                if(oneHead==null){
                    oneHead=head;
                    one=head;
                }
                else{
                    one.next=head;
                    one=head;
                }
            }
            else if(head.data==2){
                if(twoHead==null){
                    twoHead=head;
                    two=head;
                }
                else{
                    two.next=head;
                    two=head;
                }
            }
            else{
                if(zeroHead==null){
                    zeroHead=head;
                    zero=head;
                }
                else{
                    zero.next=head;
                    zero=head;
                }
            }
            head=head.next;
        }
        
         if(zero != null) zero.next = null;
        if(one != null) one.next = null;
        if(two != null) two.next = null;
        
        if(zeroHead!=null){
            if(oneHead!=null){
                zero.next=oneHead;
                one.next=twoHead;
                two=null;
            }
            else{
                if(twoHead!=null){
                    zero.next=twoHead;
                    two=null;
                }
            }
            return zeroHead;
        }
        else{
            if(oneHead==null){
             two=null;
             return twoHead;
            }    
            if(twoHead!=null){
              one.next=twoHead;
              two=null;
             }
             return oneHead;
        }
    }
}
