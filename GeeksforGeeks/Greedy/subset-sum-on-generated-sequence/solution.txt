class Solution {
    public boolean isPossible(int[] arr, int s, int x) {
        // code here
        if(x==0) return true;
        int[] nums=new int[arr.length+1];
        nums[0]=s;
        int sum=s;
        for(int i=0;i<arr.length;i++){
            nums[i+1]=sum+arr[i];
            sum+=nums[i+1];
        }
        
        for(int i=nums.length-1;i>=0;i--){
         if(x>=nums[i]){
             x-=nums[i];
         }    
        }
        return x==0;
    }
}
