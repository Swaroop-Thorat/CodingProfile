class Solution {
    public int largestRectangleArea(int[] h) {
        int max=-1,n=h.length;
        int[] L=new int[n];
        int[] R=new int[n];
        Stack<Integer> st=new Stack<>();
       
        for(int i=0;i<n;i++){
            if(!st.isEmpty()){
                if(h[st.peek()]<h[i]){
                    L[i]=st.peek();
                }
                else{
                    while(!st.isEmpty() && h[st.peek()]>=h[i]){
                        st.pop();
                    }
                    if(st.isEmpty()){
                        L[i]=-1;
                    }
                    else{
                        L[i]=st.peek();
                    }
                }
            }
            else{
                L[i]=-1;
            }

            st.push(i);
        }
 
        st.clear();

        for(int i=n-1;i>=0;i--){
            if(!st.isEmpty()){
                if(h[st.peek()]<h[i]){
                    R[i]=st.peek();
                }
                else{
                    while(!st.isEmpty() && h[st.peek()]>=h[i]){
                        st.pop();
                    }
                    if(st.isEmpty()){
                        R[i]=n;
                    }
                    else{
                        R[i]=st.peek();
                    }
                }
            }
            else{
                R[i]=n;
            }

            st.push(i);
        }

        for(int i=0;i<n;i++){
            int val=(R[i]-1)-(L[i]+1)+1;
            max=Math.max(max,val*h[i]);
        }
        return max;
    }
}
