/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public int pathSum(TreeNode root, int targetSum) {
        Map<Long,Integer> map=new HashMap<>();
        map.put(0L,1);
        return dfs(root,targetSum,0,map);
    }
    int dfs(TreeNode node,int tar,long sum,Map<Long,Integer> map){
        if(node==null) return 0;
        sum+=node.val;
        int count=0;
        if(map.containsKey(sum-tar)){
           count+=map.get(sum-tar);
        }
        map.put(sum,map.getOrDefault(sum,0)+1);
        count+=dfs(node.left,tar,sum,map);
        count+=dfs(node.right,tar,sum,map);
        int temp=map.get(sum);
        map.put(sum,temp-1);
        sum-=node.val;

        return count; 
    }
}
