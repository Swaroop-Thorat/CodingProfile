import java.util.*;
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    class Pair{
        int dist;
        TreeNode node;
        Pair(TreeNode node,int dist){
            this.dist=dist;
            this.node=node;
        }
    }
    public List<List<Integer>> verticalTraversal(TreeNode root) {
        List<List<Integer>> ans=new ArrayList<>();
        SortedMap<Integer,List<Integer>> map=new TreeMap<>();
        bfs(root,map);
        for(int key:map.keySet()){
            ans.add(map.get(key));
        }
        return ans;
    }
    void bfs(TreeNode node,SortedMap<Integer,List<Integer>> map){
        if(node==null) return;
        Queue<Pair> q=new ArrayDeque<>();
        q.offer(new Pair(node,0));
        while(!q.isEmpty()){
            int size=q.size();
            Map<Integer,List<Integer>> tempMap=new HashMap<>();
            for(int i=0;i<size;i++){
            Pair pair=q.poll();
            int dist=pair.dist;
            TreeNode curr=pair.node;
            List<Integer> temp;
            if(tempMap.containsKey(dist)){
                temp=tempMap.get(dist);
                temp.add(curr.val);
            }
            else{
                temp=new ArrayList<>();
                temp.add(curr.val);
            }
            tempMap.put(dist,temp);
            if(curr.left!=null) q.offer(new Pair(curr.left,dist-1));
            if(curr.right!=null) q.offer(new Pair(curr.right,dist+1));
         }
         for(int p:tempMap.keySet()){
            List<Integer> list=tempMap.get(p);
            Collections.sort(list);
            if(map.containsKey(p)){
               List<Integer> list2=map.get(p); 
               list2.addAll(list);
               map.put(p,list2);
            }
            else{
                map.put(p,list);
            }
         }
        }
    }
}
