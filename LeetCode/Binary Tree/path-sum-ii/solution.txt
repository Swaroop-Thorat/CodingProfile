/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public List<List<Integer>> pathSum(TreeNode root, int targetSum) {
        List<List<Integer>> ans=new ArrayList<>();
        dfs(ans,new ArrayList<>(),root,targetSum);
        return ans;
    }
    void dfs(List<List<Integer>> ans,List<Integer> list,TreeNode root,int tar){
      if(root==null) return;
      tar-=root.val;
      list.add(root.val);
      if(root.left==null && root.right==null){
        if(tar==0) ans.add(new ArrayList<>(list));
        list.remove(list.size()-1);
        return;
      }
      dfs(ans,list,root.left,tar);
      dfs(ans,list,root.right,tar);
      list.remove(list.size()-1);
    }
}
