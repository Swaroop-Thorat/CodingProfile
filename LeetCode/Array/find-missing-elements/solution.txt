class Solution {
    public List<Integer> findMissingElements(int[] nums) {
        int n=nums.length;
        List<Integer> list=new ArrayList<>();
        Arrays.sort(nums);
        for(int i=nums[0]+1;i<nums[n-1];i++){
            boolean present=false;
            for(int j=1;j<n-1;j++){
               if(nums[j]==i){
                present=true;
                continue;
               }
            }
            if(!present) list.add(i);
        }
        return list;
    }
}
