class Solution {
    public int smallestNumber(int n, int t) {
        while(!solve(n,t)){
            n++;
        }
        return n;
    }
    public boolean solve(int num,int t){
        int res=1;
        while(num>0){
            res*=num%10;
            num/=10;
            if(res==0){
                break;
            }
        }
        return res%t==0;
    }
}
