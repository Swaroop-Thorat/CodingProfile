class Solution {
    public boolean isPalindrome(int x) {
        if(x<0) return false;
        if(x==0) return true;
        int y=x;
        int res=0;
        while(x>0){
        int rem=x%10;
        res=rem+(res*10);
        x=x/10;
        }

        if(res==y) return true;
     return false;
    }
}
