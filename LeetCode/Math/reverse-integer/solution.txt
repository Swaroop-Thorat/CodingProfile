class Solution {
    public int reverse(int x) {
        boolean neg=false;
        if(x<0) neg=true;
        x=Math.abs(x);
        long y=0;
        while(x>0){
            y=(y*10)+(x%10);
            x/=10;
        }
        if(y>Integer.MAX_VALUE || y<Integer.MIN_VALUE) return 0;
        return (neg)?(int)(y*-1):(int)y;
    }
}
