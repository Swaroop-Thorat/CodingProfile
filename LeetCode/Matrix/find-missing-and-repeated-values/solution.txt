class Solution {
    public int[] findMissingAndRepeatedValues(int[][] grid) {
        int n=grid.length;
        boolean[] arr=new boolean[n*n];
        int[] res=new int[2];
        for(int i=0;i<n;i++){
            for(int j=0;j<n;j++){
                int a=grid[i][j];
                if(arr[a-1]==true){
                 res[0]=a;
                }
                arr[a-1]=true;
            }
        }

        for(int i=0;i<n*n;i++){
            if(arr[i]==false){
                res[1]=i+1;
            }
        }
        return res;
    }
}
